How to connect your own Google Sheet
  1. Create a Google Sheet with a tab named Marks and a header row: Name | Roll | Subject | Marks.
  2. In the sheet, open Extensions → Apps Script.
  3. Delete the default code and paste the script below.
  4. Save, then Deploy → New deployment → Web app.
  5. Set Execute as: Me and Who has access: Anyone, then deploy.
  6. Copy the Web app URL, paste it in the box above and click Connect.

Apps Script code (paste into the script editor):

function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Marks');
  var values = sheet.getDataRange().getValues();
  var rows = values.slice(1).map(function(r) {
    return { name: r[0], roll: r[1], subject: r[2], marks: r[3] };
  });
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Marks');
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'add') {
    sheet.appendRow([data.row.name, data.row.roll, data.row.subject, data.row.marks]);
  } else if (data.action === 'delete') {
    var values = sheet.getDataRange().getValues();
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][1]) === String(data.roll)) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
